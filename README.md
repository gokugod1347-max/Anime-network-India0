<script>
(function () {
  const container = document.getElementById('anime-updates-list');
  if (!container) return;

  fetch('updates.json', { cache: 'no-store' })
    .then(function (response) {
      if (!response.ok) throw new Error('Could not load updates.json');
      return response.json();
    })
    .then(function (updates) {
      if (!Array.isArray(updates) || updates.length === 0) {
        container.innerHTML = '<p class="updates-empty">No anime updates available.</p>';
        return;
      }

      container.innerHTML = updates.map(function (item) {
        const images = Array.isArray(item.images) ? item.images : [];
        const gallery = images.map(function (src) {
          return '<img src="' + escapeHtml(src) + '" alt="' + escapeHtml(item.title || 'Anime update') + '">';
        }).join('');

        const details = [
          item.date ? '<div><strong>📅 Date</strong><span>' + escapeHtml(item.date) + '</span></div>' : '',
          item.time ? '<div><strong>🕐 Time</strong><span>' + escapeHtml(item.time) + '</span></div>' : '',
          item.location ? '<div><strong>📍 Location</strong><span>' + escapeHtml(item.location) + '</span></div>' : ''
        ].join('');

        return '<article class="update-card">' +
          '<div class="update-gallery">' + gallery + '</div>' +
          '<div class="update-content">' +
            '<span class="update-tag">' + escapeHtml(item.tag || 'NEWS') + '</span>' +
            '<h3>' + escapeHtml(item.title || '') + '</h3>' +
            '<p class="update-summary">' + escapeHtml(item.summary || '') + '</p>' +
            (details ? '<div class="event-details">' + details + '</div>' : '') +
            (item.note ? '<p class="update-note">' + escapeHtml(item.note) + '</p>' : '') +
            (item.link ? '<a class="btn ghost update-link" href="' + escapeHtml(item.link) + '" target="_blank" rel="noopener noreferrer">More info</a>' : '') +
          '</div>' +
        '</article>';
      }).join('');
    })
    .catch(function () {
      container.innerHTML = '<p class="updates-empty">Unable to load anime updates. Please check updates.json.</p>';
    });

  function escapeHtml(value) {
    return String(value).replace(/[&<>"']/g, function (char) {
      return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#039;'}[char];
    });
  }
})();
</script>

function filterAnime(){
  const q=document.getElementById("search").value.toLowerCase().trim();
  let count=0;
  document.querySelectorAll(".update-row").forEach(row=>{
    const ok=row.dataset.title.toLowerCase().includes(q);
    row.style.display=ok?"":"none";
    if(ok) count++;
  });
  document.getElementById("noResults").hidden=count!==0;
}
